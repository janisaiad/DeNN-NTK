\section{Goal: Scaling of the NTK Correction}

\begin{frame}{The Late-Time NTK Correction}
Our goal is to understand the scaling of the first-order, late-time correction to the NTK, denoted $\Theta^{(1)}_\infty$.
\begin{theorem}[Late-Time Correction \cite{large-width-feynman}]
\begin{align*}
\Theta^{(1)}_\infty(\vec{x}) = & \underbrace{-\sum_{i=1}^N \frac{1}{\lambda_{i}}(O_{3}(\vec{x};0)\cdot\hat{e}_{i})(\Delta f_{0}\cdot\hat{e}_{i})}_{T_3} \\
& + \underbrace{\sum_{i,j=1}^N\frac{1}{\lambda_{i}(\lambda_{i}+\lambda_{j})}(\hat{e}_{i}^{T}O_{4}(\vec{x};0)\hat{e}_{j})(\Delta f_{0}\cdot \hat{e}_{i})(\Delta f_{0}\cdot \hat{e}_{j})}_{T_4}
\end{align*}
\end{theorem}
\vfill

\end{frame}

\begin{frame}{The NTK Spectrum: A Dichotomy}
    \begin{center}
        To understand the overall scaling, we must analyze the scaling of its components, $T_3$ and $T_4$.
    \end{center}

The analysis hinges on the empirical spectral structure of the NTK matrix $\Theta_0$:
\begin{itemize}
    \item \textbf{A single, isolated eigenvalue $\lambda_1$}:
    \begin{itemize}
        \item Associated eigenvector $\hat{e}_1 \approx \frac{1}{\sqrt{N}}(1, \dots, 1)^T$ (the constant mode).
        \item Scales as $\lambda_1 \sim \Order(L)$.
    \end{itemize}
    \item \textbf{A "bulk" of $N-1$ eigenvalues $\{\lambda_i\}_{i=2}^N$}:
    \begin{itemize}
        \item Associated eigenvectors $\{\hat{e}_i\}_{i=2}^N$ behave as random vectors.
        \item Scale as $\lambda_i \sim \Order(L/N)$.
    \end{itemize}
\end{itemize}
\vspace{1cm}
We use this split to analyze the sums in $T_3$ and $T_4$.
\end{frame}

\begin{frame}{The Eigenvector of Norms: A Mathematical View}
\textbf{The NTK's Bi-Homogeneity:} Due to ReLU activations, the NTK has a key scaling property with respect to its two entries:
\[ K(\lambda x_i, \mu x_j) = \lambda \mu K(x_i, x_j) \]
\vfill
\textbf{Matrix Decomposition:} This allows us to decompose the Gram matrix $K$. Let $D$ be the diagonal matrix of norms, $D_{ii}=\|x_i\|$. Then:
\[ K = D K_u D \]
where $K_u$ is the kernel of the normalized data $x_i/\|x_i\|$.
\vfill
\begin{alertblock}{The Consequence}
The principal eigenvector of this composite matrix $K$ is mathematically constrained to be aligned with $D\mathbf{1}$, which is precisely the vector of data norms.
\end{alertblock}
\end{frame}

\section{Scaling Analysis of Correction Terms}

\begin{frame}{Analysis of the $O_3$ Term}
We decompose the sum for $T_3$ into the constant mode and the bulk:
\[
T_3 = -\frac{1}{\lambda_{1}}(O_{3}\cdot\hat{e}_{1})(\Delta f_{0}\cdot\hat{e}_{1}) - \sum_{i=2}^N\frac{1}{\lambda_{i}}(O_{3}\cdot\hat{e}_{i})(\Delta f_{0}\cdot\hat{e}_{i})
\]
\vspace{0.5cm}
Substituting the eigenvalue scalings:
\[
T_3 \approx -\frac{1}{C_1 L}(O_{3}\cdot\hat{e}_{1})(\Delta f_{0}\cdot\hat{e}_{1}) - \sum_{i=2}^N\frac{N}{C_2 L}(O_{3}\cdot\hat{e}_{i})(\Delta f_{0}\cdot\hat{e}_{i})
\]
\vspace{0.5cm}
Factoring out terms from the sum reveals the dependency on $N$:
\[
T_3 \approx -\frac{1}{C_1 L}(O_{3}\cdot\hat{e}_{1})(\Delta f_{0}\cdot\hat{e}_{1}) - \frac{N(N-1)}{C_2 L} \left( \frac{1}{N-1} \sum_{i=2}^N (O_{3}\cdot\hat{e}_{i})(\Delta f_{0}\cdot\hat{e}_{i}) \right)
\]
\begin{alertblock}{Scaling of $T_3$}
The bulk term dominates, leading to a scaling of:
\[ T_3 \sim \Order\left(\frac{N^2}{L}\right) \]
\end{alertblock}
\end{frame}

\begin{frame}{Analysis of the $O_4$ Term}
The double sum for $T_4$ is decomposed into three parts:
\begin{enumerate}
    \item Top-Top interaction ($i=1, j=1$)
    \item Top-Bulk interaction ($i=1, j>1$ or $i>1, j=1$)
    \item \textbf{Bulk-Bulk interaction ($i,j > 1$) (Dominant)}
\end{enumerate}
\vspace{0.5cm}
The dominant term is:
\[
T_4^{\text{bulk}} = \sum_{i,j=2}^N \frac{(\hat{e}_{i}^{T}O_{4}\hat{e}_{j})(\Delta f_{0}\cdot \hat{e}_{i})(\Delta f_{0}\cdot \hat{e}_{j})}{\lambda_i(\lambda_i+\lambda_j)}
\]
\vspace{0.5cm}
Approximating $\lambda_i \approx \lambda_{\text{bulk}} \sim L/N$, the denominator scales as $(L/N)^2$.
\end{frame}

\begin{frame}{Analysis of the $O_4$ Term}
\[
T_4^{\text{bulk}} \approx \frac{1}{2(C_2 L/N)^2} \sum_{i,j=2}^N (\hat{e}_{i}^{T}O_{4}\hat{e}_{j})(\Delta f_{0}\cdot \hat{e}_{i})(\Delta f_{0}\cdot \hat{e}_{j})
\]
The sum has $(N-1)^2$ terms.
\begin{alertblock}{Naive scaling of $T_4$ if O4 is independent of N,L}
The overall scaling is the product of the pre-factor and the number of terms:
\[ T_4 \sim \Order\left(\frac{N^4}{L^2}\right) \]
\end{alertblock}
\end{frame}

\section{Reconciling Theory and Experiment}

\begin{frame}{The Scaling Puzzle}
\textbf{The Puzzle:} Our analysis gives raw scalings that grow rapidly with $N$, while experiments show a much milder, near-linear dependence.
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Theoretical Prediction}
\begin{itemize}
    \item $T_3 \sim \Order(N^2/L)$
    \item $T_4 \sim \Order(N^4/L^2)$
\end{itemize}
\column{0.5\textwidth}
\textbf{Empirical Observation}
\begin{itemize}
    \item $\|K_{emp} - K_\infty\| \sim \Order(NL)$
\end{itemize}
\end{columns}
\vfill
\textbf{The Missing Piece:}
This implies our "independence" assumption was wrong. The averaged tensor components must themselves have a strong scaling dependency on $N$.
\end{frame}

\begin{frame}{The Implication: A Missing Dependency}
For the theory to match experiments, the averaged tensor projection must decay rapidly.
\[
\mathbb{E}_{\hat{e}_i, \hat{e}_j}[(\hat{e}_{i}^{T}O_{4}\hat{e}_{j})(\Delta f_{0}\cdot \hat{e}_{i})(\Delta f_{0}\cdot \hat{e}_{j})]
\]
\vfill
For $T_4$ to be consistent, this term must scale as:
\[
\mathbb{E}[\dots] \sim \Order\left(\frac{L^3}{N^3}\right)
\]
This would correctly predict $T_4 \sim \Order\left(\frac{N^4}{L^2} \cdot \frac{L^3}{N^3}\right) = \Order(NL)$.
\end{frame}

\section{The Path to a Solution}

\begin{frame}{The Challenge of the $O_4$ Kernel}
To understand the scaling, we must understand the $O_4$ tensor.
\vfill
A direct, term-by-term derivation is theoretically possible but leads to an intractably complex expression, making analysis impossible.
\vfill
We need a smarter way to compute and analyze $O_4$.
\end{frame}

\begin{frame}{A Key Insight: $O_4$ as the NTK Hessian}
A crucial insight simplifies the problem immensely.
\vfill
The $O_4$ tensor can be expressed as the Hessian of the NTK ($K^{(2)}$) itself.
\begin{proposition}
$O_4(x_\alpha, x_\beta, x_\gamma, x_\delta) = \left( \nabla_\theta f_\gamma \right)^T \left( \nabla^2_\theta K^{(2)}_{\alpha\beta} \right) \left( \nabla_\theta f_\delta \right)$
\end{proposition}
\vfill
This makes $O_4$ computationally tractable using automatic differentiation frameworks like JAX.
\end{frame}

\section{A Deeper Look: Random Matrix Theory}

\begin{frame}{An Unexpected Discovery: The GOE Connection}
A detailed analysis of the NTK's spectral statistics revealed a deep connection with Random Matrix Theory (RMT).
\vfill
Specifically, the distribution of spacings for the $N-1$ eigenvalues of the NTK \textbf{bulk} perfectly matches the predictions for the Gaussian Orthogonal Ensemble (GOE).
\end{frame}

\begin{frame}{The GOE Fingerprint}
    \begin{figure}
        \centering
        \includegraphics[width=0.8\textwidth]{consecutive_spacing_ratio_dist_N64_D10_M20_L2.png}
        \caption{The empirical spacing ratio distribution (blue) is perfectly predicted by the GOE theoretical curve (red), a tell-tale sign of RMT statistics.}
    \end{figure}
\end{frame}

\begin{frame}{What This Means: Level Repulsion}
The GOE is the universality class for large real symmetric random matrices.
\vfill
A key feature is "level repulsion": eigenvalues are not independent and tend to repel each other.
\vfill
This is tested by the distribution of the ratio of consecutive spacings, $r_n = s_n / s_{n-1}$, where $s_n = \lambda_{n+1} - \lambda_n$.
\end{frame}

\begin{frame}{The GOE Test}
For uncorrelated eigenvalues, the distribution of $r$ would be Poissonian.
\vfill
For the GOE, it follows a very specific and characteristic distribution.
\begin{equation}
P_{\text{GOE}}(r) = \frac{27}{8} \frac{r + r^2}{(1 + r + r^2)^{5/2}}
\end{equation}
\vfill
This provides an extremely sensitive fingerprint for GOE statistics.
\end{frame}

\begin{frame}{Solving the Puzzle with RMT}
The GOE discovery provides the solid theoretical foundation we were missing.
\vfill
It gives us a rigorous statistical model for the $N-1$ bulk eigenvectors. Being orthogonal to the constant mode $\hat{e}_1$, they are constrained to live on the hyperplane where vector components sum to zero.
\vfill
We model them as being randomly distributed in this subspace, which corresponds to the Haar measure on $O(N-1)$.
\end{frame}

\begin{frame}{Weingarten Calculus: The Formal Solution}
We can now formally compute the required tensor averages.
\vfill
The expectation becomes an integral over the orthogonal group:
\begin{equation}
\mathbb{E}[\dots] = \int_{U \in O(N-1)} (u_i^T O_4 u_j) (v^T u_i) (v^T u_j) d\mu(U)
\end{equation}
\vfill
\textbf{Weingarten Calculus} is a method to compute moments of matrix variables distributed according to the Haar measure of compact groups. It provides the exact $1/N$ expansion to solve the integral, making the problem analytically and numerically tractable.
\end{frame}

\section{Broader Implications}

\begin{frame}{The Hessian-NTK Correspondence}
This RMT framework has even broader implications for the loss landscape.
\vfill
The Hessian of the loss can be decomposed as:
\begin{equation}
H_{\mu\nu} = \sum_{(x,y)}\left[\underbrace{\frac{\partial^{2}\mathcal{L}}{\partial f^{2}}\frac{\partial f}{\partial\theta^{\mu}}\frac{\partial f}{\partial\theta^{\nu}}}_{\mathcal{A}_{\mu\nu}}
+ \underbrace{\frac{\partial\mathcal{L}}{\partial f}\frac{\partial^{2}f}{\partial\theta^{\mu}\partial\theta^{\nu}}}_{\mathcal{B}_{\mu\nu}}\right]
\end{equation}
The first term, $\mathcal{A}$, is the NTK. The second, $\mathcal{B}$, is a residual term.
\end{frame}

\begin{frame}{Spectral Equivalence}
At large widths, the residual term's contribution to the spectrum vanishes.
\begin{equation}
\mathrm{Tr}(H^k) \approx \mathrm{Tr}(\mathcal{A}^k) + \mathrm{Tr}(\mathcal{B}^k) \quad \text{and} \quad \mathrm{Tr}(\mathcal{B}^k) \to 0
\end{equation}
\vfill
The spectrum of the Hessian becomes identical to the spectrum of the NTK.
\vfill
This means all our RMT findings for the NTK apply directly to the geometry of the loss function.
\end{frame}
