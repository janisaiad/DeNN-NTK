\begin{abstract}
We study the sampled Sobolev-NTK operator for general data distributions. Two sampling setups are formalized. We remove artificial 1/n prefactors from matrix definitions and derive mean/variance via exact formulas (CLT and quadrature exactness). The commutator is expressed as a double sum with precise control of terms. With NTK eigen-decay $\lambda_\ell \asymp \ell^{-d}$ and Sobolev growth $\mu_\ell \asymp \ell^{2s}$, we recover the scaling law for $\Kmat\Psmat$ and quantify non-commutation.
\end{abstract}

\section{Sampling setups and quadrature weights}

Let $(Y_{\ell,p})$ denote spherical harmonics on $\Sphere$ (degree $\ell\ge 0$, multiplicity $p=1,\dots,Z(d,\ell)$). For a sample $\{x_i\}_{i=1}^n \subset \Sphere$ and weights $\{q_i\}_{i=1}^n$ define
\[
(\mathbf{z}_{\ell,p})_i=\sqrt{q_i}\,Y_{\ell,p}(x_i),\qquad
G_{(\ell,p),(\ell',p')}=\mathbf{z}_{\ell,p}^{\top}\mathbf{z}_{\ell',p'}=\sum_{i=1}^n q_i\,Y_{\ell,p}(x_i)\,Y_{\ell',p'}(x_i).
\]

We analyze two precise setups:

- Setup A (data-driven Gaussian quadrature for weight p): nodes $x_i$ are random (i.i.d. from a distribution with density $p$ on $\Sphere$); weights $q_i=q_i(x_1,\dots,x_n)$ are computed so that the quadrature is exact on the finite-dimensional space
\[
\mathcal{V}_T=\operatorname{span}\{Y_{\ell,p}:\ 0\le \ell\le T\},\quad\text{i.e.}\quad \sum_{i=1}^n q_i\phi(x_i)=\int_{\Sphere}\phi(x)\,p(x)\,d\sigma(x)\ \ \forall\phi\in\mathcal{V}_T.
\]
Consequently, for all pairs with $\ell+\ell'\le T$,
\[
G_{(\ell,p),(\ell',p')}=\int_{\Sphere} Y_{\ell,p}(x)Y_{\ell',p'}(x)\,p(x)\,d\sigma(x),
\]
i.e., exact (no stochastic error) within the truncation window. The residual for $\ell+\ell'>T$ is a deterministic algebraic quadrature error (controlled by $T$).

- Setup B (Monte Carlo estimator for weight p): nodes $x_i\stackrel{\text{i.i.d.}}{\sim}p$ and fixed weights $q_i=\frac{1}{n}$ (no solve). Then for any indices $(\ell,p),(\ell',p')$,
\[
G_{(\ell,p),(\ell',p')}=\frac{1}{n}\sum_{i=1}^n Y_{\ell,p}(x_i)\,Y_{\ell',p'}(x_i),
\]
with exact mean and variance
\[
\E[G_{ab}]=\int_{\Sphere} Y_{a}(x)Y_{b}(x)\,p(x)\,d\sigma(x),\qquad
\operatorname{Var}(G_{ab})=\frac{1}{n}\Big(\E_p[Y_a^2Y_b^2]-\E_p[Y_aY_b]^2\Big).
\]
By CLT, $G_{ab}=\E[G_{ab}]+\mathcal{N}(0,\operatorname{Var}(G_{ab}))+\mathrm{o}_p(n^{-1/2})$. Here and below $a,b$ abbreviate multi-indices $(\ell,p)$.

In both setups, the resolvable bandwidth is chosen by exactness/aliasing considerations:
\[
\#\{\text{modes up to } \ell_{\max}\}\ \asymp\ \sum_{\ell\le \ell_{\max}} Z(d,\ell)\ \asymp\ \ell_{\max}^{\,d-1}\ \lesssim\ n
\quad\Rightarrow\quad \ell_{\max}\ \asymp\ n^{1/(d-1)}.
\]

\subsection{Fixed weights via orthogonal polynomials for weighted inner products}
Let $p$ be the target density on $\Sphere$ and let $\{\Phi_a\}_{a\in\mathcal{A}_T}$ be any basis of spherical polynomials up to total degree $T$ (e.g., $\Phi_a\equiv Y_{\ell,p}$ with $\ell\le T$). Given nodes $x_1,\dots,x_n$, fixed quadrature weights $\{q_i\}$ should be computed by matching weighted moments under the $L^2(p)$ inner product:
\begin{equation}
\sum_{i=1}^n q_i\,\Phi_a(x_i) \ =\ \int_{\Sphere}\Phi_a(x)\,p(x)\,d\sigma(x)\qquad \forall a\in\mathcal{A}_T. \label{eq:lin-moments}
\end{equation}
To reproduce orthogonality (and hence the Gram mass matrix) up to $T$, enforce bilinear constraints
\begin{equation}
\sum_{i=1}^n q_i\,\Phi_a(x_i)\Phi_b(x_i)\ =\ \int_{\Sphere}\Phi_a(x)\Phi_b(x)\,p(x)\,d\sigma(x) \qquad \forall a,b\in\mathcal{A}_T, \label{eq:bilinear-moments}
\end{equation}
which yields $G_{ab}=\int \Phi_a\Phi_b\,p\,d\sigma$ inside the window. In practice one solves \eqref{eq:lin-moments} (or \eqref{eq:bilinear-moments}) as a square/least-squares system; the solution produces nonuniform weights $q_i$ adapted to $p$. This is the spherical analogue of Gaussian quadrature for a general weight.

Remark (two setups, precise):
- Setup A (random nodes, quadrature weights): $x_i\stackrel{i.i.d.}{\sim}p$, then $q_i=q_i(x_{1:n})$ solve \eqref{eq:lin-moments}–\eqref{eq:bilinear-moments}; inside the degree window there is no stochastic error for $G$.
- Setup B (Monte Carlo): $x_i\stackrel{i.i.d.}{\sim}p$, $q_i=\frac{1}{n}$; then
\[
G_{ab}=\frac{1}{n}\sum_{i=1}^n \Phi_a(x_i)\Phi_b(x_i),\quad \E[G_{ab}]=\int \Phi_a\Phi_b\,p\,d\sigma,\quad \operatorname{Var}(G_{ab})=\frac{1}{n}\Big(\E_p[\Phi_a^2\Phi_b^2]-\E_p[\Phi_a\Phi_b]^2\Big).
\]

\subsection{Smallest eigenvalue scaling: K vs P$_s$ (precise normalizations)}
Define the $p$-weighted, symmetrized empirical kernel
\[
\widehat{\Kmat}\ :=\ Q^{1/2}\,K(X,X)\,Q^{1/2},\qquad Q=\mathrm{diag}(q_1,\dots,q_n),
\]
so training dynamics typically involve $\widehat{\Kmat}$ (not $K(X,X)$) because the loss integrates against $p$ and quadrature. Then under standard regularity (bounded, nondegenerate kernel on $\Sphere$, density $p$ bounded away from $0$ and $\infty$), the smallest eigenvalue obeys
\begin{equation}
\E[\lambda_{\min}(\widehat{\Kmat})]\ \asymp\ \frac{c(p,K)}{n},\qquad \operatorname{Var}(\lambda_{\min}(\widehat{\Kmat}))\ =\ O(n^{-2}), \label{eq:lmin-K-hat}
\end{equation}
i.e., the normalized operator has edge at $n^{-1}$. Consequently, for the unnormalized Gram $K(X,X)$ one has $\lambda_{\min}(K(X,X))\asymp c(p,K)$.

For the sampled Sobolev operator
\[
\Psmat=\sum_{\ell,p}\mu_\ell\,\mathbf{z}_{\ell,p}\mathbf{z}_{\ell,p}^\top,\qquad \mu_\ell\ \asymp\ \ell^{2s},
\]
the $s$-dependence is explicit and multiplicative mode-wise:
\begin{equation}
\mu_\ell(s)\ =\ e^{\,2s\log\ell}\quad(\ell\ge 1),\qquad
\lambda_{\max}(\Psmat)\ \asymp\ \ell_{\max}^{2s}\ \asymp\ n^{\frac{2s}{d-1}}, \qquad
\lambda_{\min}(\Psmat)\ \asymp\ 1, \label{eq:Ps-spectrum-s}
\end{equation}
where $\ell_{\max}\asymp n^{1/(d-1)}$ is the resolvable bandwidth. Thus $s$ stretches the spectrum of $P_s$ exponentially in $s$ (at fixed $\ell$), while the lower edge is $s$-independent (set by the first active degree).

\begin{remark}[Exponential scaling in s (by construction)]
Since $\mu_\ell(s)=\ell^{2s}=e^{2s\log\ell}$, the $s$-dependence of $\Psmat$ is explicit and multiplicative mode-wise. In particular,
$\|\Psmat\|\asymp \ell_{\max}^{2s}=e^{2s\log\ell_{\max}}$ and each harmonic is amplified by $e^{2s\log\ell}$. Therefore, the scaling in $s$ for $\Psmat$ and for the product $\Kmat\Psmat$ is fixed by construction; random-matrix analysis is only needed to quantify mode mixing through $G$ and edge fluctuations, not the $s$-scaling itself.
\end{remark}

\subsection{Consequence for the product K P$_s$}
With NTK decay $\lambda_\ell\asymp \ell^{-d}$ and Sobolev growth $\mu_\ell\asymp \ell^{2s}$, the diagonal (commuting) scaling is
\[
\lambda_\ell(\text{effective})\ \propto\ \lambda_\ell\,\mu_\ell\ \asymp\ \ell^{\,2s-d}.
\]
Non-commutation produces off-diagonal mixing controlled by
\[
[\Kmat,\Psmat]=\sum_{a,b}(\lambda_a\mu_b-\lambda_b\mu_a)\,G_{ab}\,\mathbf{z}_a\mathbf{z}_b^\top,
\]
with mean/variance of $G_{ab}$ given exactly above (Setup A: deterministic inside window; Setup B: unbiased, variance $n^{-1}$). Hence the product inherits the $\ell^{2s-d}$ scaling, while deviations are governed by quadrature exactness (Setup A) or CLT variance (Setup B).

\subsection{Future work: s-dependent random-kernel edge laws}
Goal: characterize the lower-edge law of the s-preconditioned kernel random matrix
\[
\widehat{\Kmat}\,\Psmat\ =\ Q^{1/2}\,\tilde K(\rho)\,Q^{1/2}\ \cdot\ \sum_{\ell,p}\mu_\ell\,\mathbf{z}_{\ell,p}\mathbf{z}_{\ell,p}^\top,
\]
where $\rho_{ij}=x_i\cdot x_j$ has distribution induced by $p$. Two targets:
- Edge scaling as a function of $s$: determine $a_s,b_s$ s.t. $(\lambda_{\min}-a_s)/b_s$ converges in law.
- Edge universality: identify whether a Tracy–Widom-type limit applies to the smallest eigenvalue for kernel matrices at the “hard/soft” edge under s-dependent preconditioning.

Approach: linearization and resolvent analysis for kernel random matrices with structured (sum of rank-1) PSD perturbations; anisotropic local laws; spiked/low-rank perturbation theory (see e.g. work of Dubach and related kernel/spiked models). This should refine \eqref{eq:lmin-K-hat} by giving limiting distributions (not only scaling) uniformly in $s$.

\section{Sampled operators and spectral scaling}

Continuous spectral laws (zonal NTK and Sobolev):
\[
\lambda_\ell\ \asymp\ \ell^{-d},\qquad \mu_\ell\ \asymp\ \ell^{2s}\quad(\ell\to\infty).
\]

Sampled matrices (no 1/n prefactors):
\[
\Kmat=\sum_{\ell,p}\lambda_\ell\,\mathbf{z}_{\ell,p}\mathbf{z}_{\ell,p}^{\top},\qquad
\Psmat=\sum_{\ell,p}\mu_\ell\,\mathbf{z}_{\ell,p}\mathbf{z}_{\ell,p}^{\top}.
\]
Nonzero eigenvalues of $\Psmat$ equal those of $\Lambda_s G$ with $(\Lambda_s)_{(\ell,p)}=\mu_\ell$:
\[
\operatorname{spec}_{\neq 0}(\Psmat)=\operatorname{spec}(\Lambda_s G).
\]
Thus the dependence on $s$ is entirely through $\mu_\ell\asymp \ell^{2s}$; mixing between modes is encoded in $G$ (exact in Setup A in the window, unbiased with variance n^{-1} in Setup B).

\section{Double-sum commutator: mean and variance control}

The commutator expands as a double sum of rank-1 matrices:
\[
[\Kmat,\Psmat]=\sum_{a,b}(\lambda_a\mu_b-\lambda_b\mu_a)\,G_{ab}\,\mathbf{z}_a\mathbf{z}_b^{\top}.
\]
Decompose $G_{ab}=M_{ab}+\Delta_{ab}$ with
\[
M_{ab}=
\begin{cases}
\int Y_aY_b\,p\,d\sigma,&\text{Setup A inside exactness window and Setup B (mean)}\\
\text{same},&\text{outside window (Setup A) but with deterministic quadrature residual}
\end{cases},
\qquad
\Delta_{ab}=
\begin{cases}
0,&\text{Setup A, if } \ell_a+\ell_b\le T\\
\text{quad. error},&\text{Setup A, else}
\end{cases}
\]
and in Setup B (Monte Carlo),
\[
\E[\Delta_{ab}]=0,\qquad \operatorname{Var}(\Delta_{ab})=\frac{1}{n}\Big(\E_p[Y_a^2Y_b^2]-\E_p[Y_aY_b]^2\Big).
\]

Hence
\[
[\Kmat,\Psmat]=\underbrace{\sum_{a,b}(\lambda_a\mu_b-\lambda_b\mu_a)\,M_{ab}\,\mathbf{z}_a\mathbf{z}_b^{\top}}_{\text{mean/nonrandom part}}
+\underbrace{\sum_{a,b}(\lambda_a\mu_b-\lambda_b\mu_a)\,\Delta_{ab}\,\mathbf{z}_a\mathbf{z}_b^{\top}}_{\text{fluctuation}}.
\]

Norm control (Frobenius) via triangle inequality and Cauchy–Schwarz:
\[
\|[\Kmat,\Psmat]\|_F\ \le\ \sum_{a,b}|\lambda_a\mu_b-\lambda_b\mu_a|\ |G_{ab}|\ \|\mathbf{z}_a\mathbf{z}_b^{\top}\|_F,
\quad \|\mathbf{z}_a\mathbf{z}_b^{\top}\|_F=\|\mathbf{z}_a\|\,\|\mathbf{z}_b\|=\sqrt{G_{aa}G_{bb}}.
\]
In Setup B (MC), exact first and second moments yield
\[
\E\big[|G_{ab}|\sqrt{G_{aa}G_{bb}}\big]\ \le\ \sqrt{\E[G_{ab}^2]}\,\sqrt[4]{\E[G_{aa}^2]\E[G_{bb}^2]},
\]
with
\[
\E[G_{ab}^2]=\operatorname{Var}(G_{ab})+M_{ab}^2
=\frac{1}{n}\Big(\E_p[Y_a^2Y_b^2]-M_{ab}^2\Big)+M_{ab}^2,\qquad
\E[G_{aa}^2]=\frac{1}{n}\Big(\E_p[Y_a^4]-\E_p[Y_a^2]^2\Big)+\E_p[Y_a^2]^2.
\]
Therefore each summand is controlled by explicit moments under $p$ and decays at rate n^{-1/2} around its mean contribution.

Counting modes up to $\ell_{\max}\asymp n^{1/(d-1)}$ gives $\#\{a\}\asymp n$, hence a double sum with $\asymp n^2$ terms. Using $\lambda_a\asymp \ell_a^{-d}$ and $\mu_a\asymp \ell_a^{2s}$,
\[
|\lambda_a\mu_b-\lambda_b\mu_a|\ \lesssim\ \ell_a^{-d}\ell_b^{2s}+\ell_b^{-d}\ell_a^{2s}.
\]
Consequently,
\[
\|[\Kmat,\Psmat]\|_F\ \lesssim\ \sum_{\ell_a,\ell_b\le \ell_{\max}}
(\ell_a^{-d}\ell_b^{2s}+\ell_b^{-d}\ell_a^{2s})\ \Big(|M_{ab}|+O(n^{-1/2})\Big)\ \sqrt{G_{aa}G_{bb}}.
\]
- Setup A: inside the exactness window ($\ell_a+\ell_b\le T$), $M_{ab}$ is deterministic; choosing Gaussian quadrature degree $T\ge 2\ell_{\max}$ makes $M_{ab}$ the exact $L^2(p)$ inner product and removes stochastic error. Outside the window, the quadrature residual controls $M_{ab}-\int Y_aY_b p$.
- Setup B: the fluctuation contribution is explicitly $\mathcal{N}(0,\operatorname{Var})$ with variance $O(n^{-1})$ per entry, yielding total fluctuation of order $n^{-1/2}$ at the level of each summand and a net scaling governed by the mode count and the weights $(\lambda,\mu)$.

In both setups, the dominant spectral scaling of $\Kmat\Psmat$ follows the diagonal weights $\lambda_\ell\mu_\ell\asymp \ell^{2s-d}$, while non-commutation is controlled by the above mean/variance terms.

\section{Resulting scaling law for $\Kmat\Psmat$}
With bandwidth $\ell_{\max}\asymp n^{1/(d-1)}$ and eigenweight law $\lambda_\ell\mu_\ell\asymp \ell^{2s-d}$,
\[
\text{per-harmonic gain:}\quad \lambda_\ell\,\mu_\ell\ \asymp\ \ell^{2s-d},\qquad
\text{sampled spectrum:}\quad \lambda_k(\Psmat)\ \asymp\ \ell_k^{2s}\ \text{modulated by }G,
\]
and the product $\Kmat\Psmat$ inherits the $\ell^{2s-d}$ scaling; commutator-induced mixing is small in Setup A (by exactness) and mean-zero with variance $O(n^{-1})$ per Gram entry in Setup B (by CLT), hence does not change the scaling law.

\section{RMT perspective (future work)}
Kernel random matrix:
\[
(\Kmat)_{ij}=K_{\mathrm{NTK}}(x_i,x_j)=\tilde K(\rho_{ij}),\qquad \rho_{ij}=x_i\cdot x_j,
\]
where the law of $\rho_{ij}$ is induced by $p$. Sobolev operator:
\[
\Psmat=\sum_{\ell,p}\mu_\ell\,\mathbf{z}_{\ell,p}\mathbf{z}_{\ell,p}^{\top}\quad\text{(PSD sum of rank-1 terms)}.
\]
Thus $\Kmat\Psmat=\sum_{\ell,p}\mu_\ell\,(\Kmat\mathbf{z}_{\ell,p})\,\mathbf{z}_{\ell,p}^{\top}$ is a structured low-/moderate-rank perturbation of $\Kmat$ (rank equal to number of retained harmonics). Rank-one perturbation theory for kernel random matrices (see e.g. Dubach and related works) can yield precise outlier locations and eigenvector overlaps, providing non-asymptotic training guarantees.
